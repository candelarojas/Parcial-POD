🎯 Objetivo del Estudio

Evaluar la evolución temporal y la distribución por especialidades de la demanda en una clínica privada de Turdera, y estimar el impacto económico del gasto de bolsillo anual para una familia tipo a partir de los aranceles de clínicas privadas del Conurbano durante 2026.

# @title Barra promedio mensual
import pandas as pd
import plotly.express as px

df = pd.read_csv("/content/drive/MyDrive/clinica_turdera.csv")

df_mensual = (
    df.groupby(['Mes', 'Consulta']).size().reset_index(name='Cantidad')
)

df_promedio = (
    df_mensual.groupby('Consulta')['Cantidad'].mean().reset_index(name='Promedio_Mensual')
)

fig = px.bar(
    df_promedio,
    x='Consulta',
    y='Promedio_Mensual',
    color='Consulta',
    color_discrete_sequence=["#FF4A5A", "#2EC4B6", "#FF9F1C", "#E71D36", "#3A86FF","#8338EC", "#FF007F", "#00F5D4", "#FFBE0B", "#9B5DE5", "#00BBF9", "#FF5400", "#70E000", "#EA526F", "#4CC9F0", "#F15BB5", "#00B4D8", "#F72585"],
    title='Promedio de Consultas Mensuales por Especialidad',
    labels={
        'Consulta': 'Especialidad',
        'Promedio_Mensual': 'Promedio de Consultas / Mes',
    },
    template = "plotly_dark",
)

fig.update_layout(xaxis={'categoryorder': 'total descending'})

fig.show()

# @title Boxplot Costos
import pandas as pd
import plotly.express as px

data_filtrada = data[data["Servicio"] != "Mensualidad Socios"]

data_long = data_filtrada.melt(
    id_vars=["Servicio"], value_vars=[col for col in data.columns if col != "Servicio"], var_name="Columna_Original", value_name="Costo"
)

def mapear_clinica_y_socio(fila):
    nombre_col = fila["Columna_Original"].strip()
    condicion = "Socio" if "Socios" in nombre_col else "No Socio"

    if "San Martin" in nombre_col:
        clinica = "Sala San Martin"
    elif "Monteagudo" in nombre_col:
        clinica = "C. M. Monteagudo"
    elif "sala salud" in nombre_col or "Sala Salud" in nombre_col:
        clinica = "C. M. Sala Salud"
    else:
        clinica = "Otra"

    return pd.Series([clinica, condicion])

data_long[["Clínica", "Condición"]] = data_long.apply(mapear_clinica_y_socio, axis=1)

fig = px.box(
    data_long,
    x="Clínica",
    y="Costo",
    color="Condición",
    hover_data=["Servicio"],
    title="Distribución de Costos Médicos por Clínica y Beneficio de Socio",
    labels={'Costo': 'Costo del Servicio ($)', 'Clínica': 'Establecimiento Médico'},
    color_discrete_sequence=["#4abec2", "#f0aa3a"]
)

fig.update_layout(
    boxmode="group",
    yaxis_tickformat=",.0f",
    template="plotly_white"
)

fig.show()
